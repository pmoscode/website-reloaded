{{- /* Responsives Bild auf feste Größe zugeschnitten (Karten, Thumbnails). dict: res (Resource), w, h, sizes, alt, class, eager (bool), priority (bool) priority = wichtigstes Bild der Seite (LCP): eager + fetchpriority="high". srcset: 1×, 1,5× und 2× – soweit das Original groß genug ist. GIF/SVG werden unverändert ausgeliefert (Animation bleibt erhalten). */ -}} {{- $r := .res -}} {{- $w := .w | default 800 -}} {{- $h := .h | default 500 -}} {{- $prio := .priority | default false -}} {{- $loading := cond (or $prio (.eager | default false)) "eager" "lazy" -}} {{- if or (not (reflect.IsImageResourceProcessable $r)) (eq $r.MediaType.SubType "gif") -}} {{- else -}} {{- $s1 := $r.Fill (printf "%dx%d webp smart" $w $h) -}} {{- $set := slice (printf "%s %dw" $s1.RelPermalink $s1.Width) -}} {{- range slice 1.5 2.0 -}} {{- $sw := int (mul $w .) -}} {{- $sh := int (mul $h .) -}} {{- if ge $r.Width $sw -}} {{- $s := $r.Fill (printf "%dx%d webp smart" $sw $sh) -}} {{- $set = $set | append (printf "%s %dw" $s.RelPermalink $s.Width) -}} {{- end -}} {{- end -}} {{- end -}}
