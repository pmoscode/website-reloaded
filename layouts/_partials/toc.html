{{- /* Inhaltsverzeichnis aus den Überschriften – unabhängig davon, ob ein Artikel mit # oder ## beginnt. Zeigt die oberste vorhandene Ebene und eine darunter. Gibt "" zurück, wenn es weniger als 3 Einträge gibt. */ -}} {{- $hm := .Fragments.HeadingsMap -}} {{- $hs := slice -}} {{- $min := 9 -}} {{- range .Fragments.Identifiers -}} {{- with index $hm . -}} {{- if and .Title (le .Level 4) -}} {{- $hs = $hs | append . -}} {{- if lt .Level $min }}{{ $min = .Level }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- $items := slice -}} {{- range $hs }}{{ if le .Level (add $min 1) }}{{ $items = $items | append . }}{{ end }}{{ end -}} {{- $out := "" -}} {{- if ge (len $items) 3 -}} {{- $out = "
    " -}} {{- range $items -}} {{- $out = printf `%s
  1. %s
    2. ` $out (sub .Level $min) .ID (.Title | plainify) -}} {{- end -}} {{- $out = printf "%s
" $out -}} {{- end -}} {{- return ($out | safeHTML) -}}
